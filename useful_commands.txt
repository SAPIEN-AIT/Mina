    cd ~/dev/Mina && docker compose -f docker/docker-compose.yaml build isaac-lab-template

    docker compose -f docker/docker-compose.yaml run --rm isaac-lab-template -c \
      '/.venv/bin/python3 --version && readlink -f /.venv/bin/python3 && id && echo SMOKE_OK'

    docker compose -f docker/docker-compose.yaml run --rm isaac-lab-template -c '
      set -e
      touch /root/.cache/ov/test
      touch /root/.cache/pip/test
      touch /root/.cache/nvidia/GLCache/test
      touch /root/.nv/ComputeCache/test
      touch /root/.nvidia-omniverse/logs/test
      touch /root/.local/share/ov/data/test
      touch /root/Documents/test
      echo WRITE_OK'

GPU check:
    docker compose -f docker/docker-compose.yaml run --rm isaac-lab-template -c \
      '/.venv/bin/python3 -c "import torch; print(f\"CUDA: {torch.cuda.is_available()}, devices: {torch.cuda.device_count()}\")"'
